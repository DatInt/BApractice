```
@startuml
' Information Engineering notation
skinparam linetype ortho
skinparam roundcorner 10

entity Customer {
  +id : bigint 
  --
  first_name : varchar
  last_name : varchar
  email : varchar
  phone : varchar
  created_at : datetime
}

entity Seller {
  +id : bigint 
  --
  company_name : varchar
  contact_name : varchar
  email : varchar
  phone : varchar
  created_at : datetime
}

entity "Order" as Order {
  +id : bigint 
  --
  order_date : datetime
  status : varchar
  total_amount : decimal
  customer_id : bigint <<FK>>
}

entity OrderItem {
  +id : bigint 
  --
  quantity : int
  price : decimal
  order_id : bigint <<FK>>
  product_id : bigint <<FK>>
}

entity Product {
  +id : bigint 
  --
  name : varchar
  description : text
  price : decimal
  stock_qty : int
  category_id : bigint <<FK>>
  seller_id : bigint <<FK>>
}

entity Category {
  +id : bigint 
  --
  name : varchar
  description : text
}

Customer ||--o{ Order : places
Order ||--|{ OrderItem : contains
Product ||--o{ OrderItem : included_in
Category ||--o{ Product : categorizes
Seller ||--o{ Product : sells

@enduml
```

## Описание сущностей БД
### Customer (Покупатель)

| Название   | Описание                            | Обязательность | Формат данных |
| ---------- | ----------------------------------- | -------------- | ------------- |
| id         | Уникальный идентификатор покупателя | Да             | bigint        |
| first_name | Имя покупателя                      | Да             | varchar(100)  |
| last_name  | Фамилия покупателя                  | Да             | varchar(100)  |
| email      | Email для входа и уведомлений       | Да             | varchar(255)  |
| phone      | Контактный телефон                  | Нет            | varchar(20)   |
| created_at | Дата регистрации пользователя       | Да             | datetime      |
### Seller (Продавец)

|Название|Описание|Обязательность|Формат данных|
|---|---|---|---|
|id|Уникальный идентификатор продавца|Да|bigint|
|company_name|Название компании продавца|Да|varchar(255)|
|contact_name|Контактное лицо|Нет|varchar(255)|
|email|Email для связи с продавцом|Да|varchar(255)|
|phone|Телефон продавца|Нет|varchar(20)|
|created_at|Дата регистрации продавца|Да|datetime|
### Product (Товар)

| Название    | Описание                        | Обязательность | Формат данных |
| ----------- | ------------------------------- | -------------- | ------------- |
| id          | Уникальный идентификатор товара | Да             | bigint        |
| name        | Название товара                 | Да             | varchar(255)  |
| description | Описание товара                 | Нет            | text          |
| price       | Цена товара                     | Да             | decimal(10,2) |
| stock_qty   | Количество товара на складе     | Да             | int           |
| category_id | Идентификатор категории товара  | Да             | bigint        |
| seller_id   | Идентификатор продавца          | Да             | bigint        |
### Category (Категория)

|Название|Описание|Обязательность|Формат данных|
|---|---|---|---|
|id|Уникальный идентификатор категории|Да|bigint|
|name|Название категории|Да|varchar(255)|
|description|Описание категории|Нет|text|
### Order (Заказ)

|Название|Описание|Обязательность|Формат данных|
|---|---|---|---|
|id|Уникальный идентификатор заказа|Да|bigint|
|order_date|Дата оформления заказа|Да|datetime|
|status|Статус заказа (создан, оплачен, отправлен и т.д.)|Да|varchar(50)|
|total_amount|Общая сумма заказа|Да|decimal(12,2)|
|customer_id|Идентификатор покупателя|Да|bigint|
### OrderItem (Позиция заказа)

| Название   | Описание                                | Обязательность | Формат данных |
| ---------- | --------------------------------------- | -------------- | ------------- |
| id         | Уникальный идентификатор позиции заказа | Да             | bigint        |
| order_id   | Идентификатор заказа                    | Да             | bigint        |
| product_id | Идентификатор товара                    | Да             | bigint        |
| quantity   | Количество товара в заказе              | Да             | int           |
| price      | Цена товара на момент заказа            | Да             | decimal(10,2) |
